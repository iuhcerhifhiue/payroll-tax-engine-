Reference

          
POST /api/paycheck

          

            One endpoint. Amounts in integer cents, every tax line reporting the taxable base it was computed
            on. The sandbox above sends exactly this.
          

          

            

              
Request


// Authorization: Bearer sk_test_…
{
  "checkDate": "2026-06-15",
  "payFrequency": "biweekly",
  "earnings": [
    { "code": "REG", "category": "regular", "amount": 300000 }
  ],
  "deductions": [
    { "code": "401K", "category": "deferral_401k", "amount": 24000 }
  ],
  "federalW4": { "filingStatus": "single" },
  "ytd": { "socialSecurity": 0, "medicare": 0, "futa": 0 },
  "workState": { "code": "PA" }
}
